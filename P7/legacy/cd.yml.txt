# .github/workflows/ (y se le cambio la extension a .txt) a proposito: es
# el workflow de despliegue directo de P7, que la Practica 8 prohibe
# explicitamente que exista dentro de .github/workflows del repositorio.
# Se conserva aqui solo como referencia historica de lo que hacia P7 antes
# de migrar a GitOps (ver P8/docs/GITOPS.md).

name: CD (legacy P7 - deshabilitado por defecto)

# Practica 7 (Software Avanzado, USAC): Despliegue Continuo del chart de
# Helm sa-platform (P5/charts/sa-platform) hacia un cluster de Kubernetes.
#
# SUPERSEDIDO POR LA PRACTICA 8 (GitOps): este workflow ejecutaba
# `helm upgrade --install` directamente contra el cluster desde GitHub
# Actions, exactamente lo que P8 prohibe explicitamente (ningun workflow
# debe correr kubectl/helm/terraform contra el cluster; ArgoCD es el unico
# componente que aplica cambios, ver P8/docs/GITOPS.md). Se mantiene este
# archivo por referencia historica de P7 y como fallback manual local, pero
# su disparador automatico (workflow_run tras CI) quedo deshabilitado: solo
# se ejecuta si alguien lo lanza a mano con "workflow_dispatch". El flujo
# vigente para publicar una version es .github/workflows/gitops-update.yml
# (build + Trivy + SBOM + cosign + Pull Request al repositorio GitOps).
#
# LIMITACION DOCUMENTADA (ver P7/README.md "Kubernetes" y "Problemas
# conocidos"): el cluster de GKE usado en la Practica 6 fue eliminado
# deliberadamente para no seguir generando costo. Un runner alojado por
# GitHub (en la nube) no puede alcanzar un cluster que vive en la
# maquina local del estudiante (Docker Desktop Kubernetes / minikube).
# Por eso este job corria en un runner "self-hosted": la propia maquina
# del estudiante, que ya tiene configurado su contexto de kubectl y
# Helm instalado (igual que en P5/P6). Ver P7/README.md seccion
# "Self-hosted runner" para el procedimiento de registro.
on:
  workflow_dispatch: {}

jobs:
  deploy:
    name: Helm upgrade + verificacion de rollout (manual, legacy)
    runs-on: [self-hosted]
    defaults:
      run:
        shell: bash
    env:
      REGISTRY: ghcr.io
      IMAGE_NAMESPACE: sa-platform
      NAMESPACE: sa-p5
      RELEASE: sa-platform
      IMAGE_TAG: ${{ github.sha }}
    steps:
      - uses: actions/checkout@v4

      - name: Verificar herramientas disponibles en el runner
        run: |
          helm version
          kubectl version --client
          kubectl config current-context

      - name: Verificar que el archivo de secretos locales existe
        run: |
          if [ -z "${SECRETS_PATH}" ]; then
            echo "::error::La variable de repositorio SA_PLATFORM_SECRETS_PATH no esta configurada."
            echo "Debe apuntar, en la maquina del self-hosted runner, a una copia de"
            echo "P5/charts/sa-platform/values-secrets.yaml (NUNCA versionada en git)."
            exit 1
          fi
          if [ ! -f "${SECRETS_PATH}" ]; then
            echo "::error::No se encontro el archivo de secretos en: ${SECRETS_PATH}"
            exit 1
          fi
        env:
          SECRETS_PATH: ${{ vars.SA_PLATFORM_SECRETS_PATH }}

      - name: helm dependency build (postgresql + rabbitmq)
        working-directory: P5/charts/sa-platform
        run: |
          helm repo add bitnami https://charts.bitnami.com/bitnami --force-update
          helm dependency build .

      - name: helm upgrade --install (imagenes con tag = commit SHA)
        working-directory: P5/charts/sa-platform
        run: |
          helm upgrade --install "${RELEASE}" . \
            --namespace "${NAMESPACE}" --create-namespace \
            -f values-dev.yaml \
            -f "${{ github.workspace }}/P7/helm/values-ci.yaml" \
            -f "${{ vars.SA_PLATFORM_SECRETS_PATH }}" \
            --set gateway.image.tag="${IMAGE_TAG}" \
            --set ms-users.image.tag="${IMAGE_TAG}" \
            --set ms-products.image.tag="${IMAGE_TAG}" \
            --set ms-orders.image.tag="${IMAGE_TAG}" \
            --set ms-notifications.image.tag="${IMAGE_TAG}" \
            --set cronJobs.heartbeat.image.tag="${IMAGE_TAG}" \
            --set cronJobs.summary.image.tag="${IMAGE_TAG}" \
            --wait --timeout 10m

      - name: Verificar rollout de cada Deployment
        run: |
          for deploy in gateway ms-users ms-products ms-orders ms-notifications; do
            echo "== kubectl rollout status deployment/${deploy} =="
            kubectl rollout status "deployment/${deploy}" -n "${NAMESPACE}" --timeout=180s
          done

      - name: Estado final de pods y cronjobs (evidencia)
        if: always()
        run: |
          kubectl get deployments -n "${NAMESPACE}" -o wide
          kubectl get pods -n "${NAMESPACE}" -o wide
          kubectl get cronjobs -n "${NAMESPACE}"
